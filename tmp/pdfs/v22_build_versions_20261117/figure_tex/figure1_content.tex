% Figure 1 is 7 x 4.25 inches. One coordinate unit equals 1/400 inch.
% Each PNG is one transparent square model tile. The layout, labels, legend,
% arrows, and separators are editable TikZ elements.
\begingroup%
\newdimen\fOneUnit%
\fOneUnit=.0025in\relax%
\newcommand{\fOneTile}[4]{%
  \node at (#2,#3) {%
    \includegraphics[width=\dimexpr\fOneUnit*#4\relax]{figures/figure1_models/#1.png}};%
}%
% Left edge, top edge, tile edge, and categories in reading order.
\newcommand{\fOneRow}[4]{%
  \pgfmathtruncatemacro{\tileY}{#2+#3/2}%
  \foreach \category [count=\groupIndex from 0] in {#4} {%
    \pgfmathtruncatemacro{\leftX}{#1+\groupIndex*(3*#3+36)+#3/2}%
    \pgfmathtruncatemacro{\middleX}{\leftX+#3+8}%
    \pgfmathtruncatemacro{\rightX}{\middleX+#3+8}%
    \fOneTile{\category-voxel}{\leftX}{\tileY}{#3}%
    \fOneTile{\category-refined}{\middleX}{\tileY}{#3}%
    \fOneTile{\category-brickforma}{\rightX}{\tileY}{#3}%
    \ifnum\groupIndex>0
      \draw[soft divider]
        ({\leftX-#3/2-10},{#2+8}) -- ({\leftX-#3/2-10},{#2+#3-8});
    \fi
  }%
}%
