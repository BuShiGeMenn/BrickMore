python3 -c 'from pathlib import Path; import hashlib; names=("setup.sh","run.sh","Scripts/environment.sh"); p=Path("CHECKSUMS.sha256"); pairs=[s.split("  ",1) for s in p.read_text(encoding="utf-8-sig").splitlines()]; expected={n:h for h,n in pairs}; old={n:Path(n).read_bytes() for n in names}; assert all(hashlib.sha256(b).hexdigest()==expected[n] for n,b in old.items()),"Script checksum mismatch"; fixed={n:b.replace(b"\r\n",b"\n") for n,b in old.items()}; [Path(n).write_bytes(b) for n,b in fixed.items()]; p.write_bytes("".join((hashlib.sha256(fixed[n]).hexdigest() if n in fixed else h)+"  "+n+"\n" for h,n in pairs).encode()); print("LF_REPAIR_OK")'

bash setup.sh --offline
